    public function login($email, $password) {
            $query = "SELECT * FROM user WHERE email = '$email'";
            $result = $this->db->query($query);
        
            if ($result->num_rows > 0) {
                $user = $result->fetch_assoc();
        
                // Debug: Tampilkan data yang diambil dari database
                echo "Password (Database): " . $user['password'] . "<br>";
                echo "Password (Input): " . $password . "<br>";
        
                if (password_verify($password, $user['password'])) { // Verify password
                    $this->user_id = $user['user_id'];
                    $this->role = $user['role']; // Fetch user role from database
                    return true;
                } else {
                    echo "Password verification failed!<br>"; // Debug if password verification fails
                }
            } else {
                echo "No user found with this email!<br>"; // Debug if no user is found
            }
            return false;
    }
